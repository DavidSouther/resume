-- Question: For each customer, find all products from their last 10 purchases.
-- Query:
with ranked_purchases as (select purchaseid, customerid, purchasetimestamp, row_number() over (partition by customerid order by purchasetimestamp desc) as rn from purchase) select c.customerid, pi.productid, rp.purchasetimestamp from customer as c join ranked_purchases as rp on rp.customerid = c.customerid and rp.rn <= 10 join purchase_items as pi on pi.purchaseid = rp.purchaseid;
