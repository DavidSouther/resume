-- Question: What percentage of customers have ever purchased productid=42?
-- Query:
with purchase_42 as (select distinct c.customerid from customer as c join purchase as p on c.customerid = p.customerid join purchase_items as pi on p.purchaseid = pi.purchaseid where pi.productid = 42) select 100.0 * count(distinct p42.customerid) / count(distinct c.customerid) as pct from customer as c left join purchase_42 as p42 on p42.customerid = c.customerid;
