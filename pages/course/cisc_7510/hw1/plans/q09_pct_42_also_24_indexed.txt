-- Question: Of customers who purchased productid=42, what percentage also purchased productid=24?
-- Query:
with purchased_42 as (select distinct p.customerid from purchase as p join purchase_items as pi on pi.purchaseid = p.purchaseid where pi.productid = 42), purchased_24 as (select distinct p.customerid from purchase as p join purchase_items as pi on pi.purchaseid = p.purchaseid where pi.productid = 24) select 100.0 * count(distinct p24.customerid) / count(distinct p42.customerid) as pct from purchased_42 as p42 left join purchased_24 as p24 on p24.customerid = p42.customerid;
