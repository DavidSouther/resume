-- Question: What is the most popular (purchased most often) product in Tri-state Area? (NJ, NY, CT)
-- Query:
select pi.productid, count(*) as purchases from purchase_items as pi join purchase as p on p.purchaseid = pi.purchaseid join customer as c on c.customerid = p.customerid where c.state in ('NJ', 'NY', 'CT') group by pi.productid order by purchases desc limit 1;
