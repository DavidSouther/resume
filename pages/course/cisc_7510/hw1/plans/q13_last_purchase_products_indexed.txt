-- Question: For each customer, find all products from their last purchase.
-- Query:
with last_purchase as (select customerid, max(purchasetimestamp) as last_ts from purchase group by customerid) select c.customerid, pi.productid from customer as c join last_purchase as lp on lp.customerid = c.customerid join purchase as p on p.customerid = c.customerid and p.purchasetimestamp = lp.last_ts join purchase_items as pi on pi.purchaseid = p.purchaseid;
