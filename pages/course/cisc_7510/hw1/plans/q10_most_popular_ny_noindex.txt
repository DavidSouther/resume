-- Question: What is the most popular (purchased most often) product in NY state?
-- Query:
select pi.productid, count(*) as purchases from purchase_items as pi join purchase as p on p.purchaseid = pi.purchaseid join customer as c on c.customerid = p.customerid where c.state = 'NY' group by pi.productid order by purchases desc limit 1;
